cmake_minimum_required(VERSION 3.20)
project(PAFVideos VERSION 0.1.1 LANGUAGES CXX)
set(PAF_RELEASE_VERSION "0.1.1")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE src)
add_test(NAME core COMMAND core_test)
if(WIN32)
  enable_language(RC)
  configure_file(src/version.h.in generated/version.h @ONLY)
  add_executable(PAFVideos WIN32 src/main.cpp src/updater.cpp assets/app.rc)
  add_executable(PAFVideosUpdater WIN32 src/update_helper.cpp src/updater.cpp)
  set_target_properties(PAFVideos PROPERTIES OUTPUT_NAME "PAF-Videos")
  set_target_properties(PAFVideosUpdater PROPERTIES OUTPUT_NAME "PAF-Videos.Update")
  foreach(target PAFVideos PAFVideosUpdater)
    target_include_directories(${target} PRIVATE src assets "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(${target} PRIVATE winhttp bcrypt advapi32 shell32 ole32)
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8 /W4)
    elseif(MINGW)
      target_link_options(${target} PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
  endforeach()
  target_link_libraries(PAFVideos PRIVATE gdiplus comctl32 comdlg32 shlwapi dwmapi uxtheme uuid)
  if(MSVC)
    target_link_options(PAFVideos PRIVATE /MANIFEST:NO)
  endif()
endif()
